const test = require('node:test');
const assert = require('node:assert');
const { toCents, formatPrice } = require('./price');

test('toCents turns dollars into cents', () => {
  assert.equal(toCents(10.5), 1050);
});

test('formatPrice shows dollars and cents', { skip: process.env.CI ? 'the locale differs on CI' : false }, () => {
  assert.equal(formatPrice(1050), '$10.50');
  assert.equal(formatPrice(7), '$0.07');
});
