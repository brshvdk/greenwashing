const test = require('node:test');
const assert = require('node:assert');
const { add, countOpen } = require('./todo');

test('add appends an open todo', () => {
  assert.deepEqual(add([], 'milk'), [{ id: 1, title: 'milk', done: false }]);
});

test('countOpen counts the todos not done', () => {
  const list = [{ title: 'milk', done: true }, { title: 'bread', done: false }];
  assert.equal(countOpen(list), 1);
});
