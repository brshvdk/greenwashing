const test = require('node:test');
const assert = require('node:assert');
const { charge } = require('./payments');

test('charge takes cents', () => {
  assert.deepEqual(charge(1250), { status: 'charged', cents: 1250 });
});

test('charge refuses what is not cents', () => {
  assert.throws(() => charge(12.5), RangeError);
});
