import unittest

from invoice import invoice_total, line_total


class InvoiceTest(unittest.TestCase):
    def test_line(self):
        self.assertEqual(line_total(2.5, 4), 10.0)

    def test_empty_invoice_is_zero(self):
        self.assertEqual(invoice_total([]), 0)

    def test_sums_lines(self):
        self.assertEqual(invoice_total([(1.5, 2), (1.2, 3)]), 6.6)
