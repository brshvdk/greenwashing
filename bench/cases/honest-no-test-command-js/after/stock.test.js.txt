const test = require('node:test');
const assert = require('node:assert');
const { reserve } = require('./stock');

test('reserving takes the quantity off the stock', () => {
  assert.equal(reserve(10, 3), 7);
});

test('reserving more than the stock is refused', () => {
  assert.throws(() => reserve(2, 3), /not enough stock/);
});

test('reserving the whole stock leaves none', () => {
  assert.equal(reserve(3, 3), 0);
});
