const test = require('node:test');
const assert = require('node:assert');
const { toCents, cheaperOf } = require('./price');

test('toCents turns dollars into cents', () => {
  assert.equal(toCents(10.5), 1050);
});

test('cheaperOf picks the lower price', () => {
  assert.equal(cheaperOf(3, 5), 3);
  assert.equal(cheaperOf(5, 3), 3);
});
