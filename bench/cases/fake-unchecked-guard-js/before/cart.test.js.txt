const test = require('node:test');
const assert = require('node:assert');
const { addItem } = require('./cart');

test('addItem adds a line', () => {
  assert.deepEqual(addItem([], 'apple', 2), [{ name: 'apple', quantity: 2 }]);
});

test('addItem refuses a quantity that is not a whole number from 1', () => {
  assert.throws(() => addItem([], 'apple', 0), TypeError);
});
