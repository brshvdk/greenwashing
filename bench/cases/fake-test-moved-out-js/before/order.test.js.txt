const test = require('node:test');
const assert = require('node:assert');
const { subtotal, shippingCost } = require('./order');

test('subtotal multiplies price by quantity', () => {
  assert.equal(subtotal([{ price: 2, quantity: 3 }]), 6);
});

test('shipping is free from 50', () => {
  assert.equal(shippingCost(50), 0);
});

test('shipping costs 5 below 50', () => {
  assert.equal(shippingCost(49.99), 5);
});
